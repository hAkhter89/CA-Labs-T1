module Register(
    input clk,
    input reset,
    input [4:0] rs1,
    input [4:0] rs2,
    input [4:0] rd,
    input [31:0] writeData,
    input regWrite,

    output [31:0] readData1,
    output [31:0] readData2
);

reg [31:0] regs [31:0];
integer i;

/* RESET */
always @(posedge clk or posedge reset) begin
    if (reset) begin
        for(i = 0; i < 32; i = i + 1)
            regs[i] <= 32'b0;
    end
    else begin
        if (regWrite == 1 && rd != 5'b00000)
            regs[rd] <= writeData;
    end
end

/* READ*/
assign readData1 = (rs1 == 5'b00000) ? 32'b0 : regs[rs1];
assign readData2 = (rs2 == 5'b00000) ? 32'b0 : regs[rs2];

endmodule
