main: 
    addi x10, x0, 0         # state = INPUT_WAITING 
    addi x2,  x0, 5         # x2 = 5 
    sw   x2,  4(x0)         # store 5 into switch_port (memory[4]) 

main_loop: 
    lw   x1,  0(x0)         # load reset_flag 
    addi x3,  x0, 1         # x3 = 1 
    beq  x1,  x3, reset     # if reset_flag == 1, goto reset 
    beq  x10, x0, input     # if state == 0, goto input 

countdown: 
    beq  x2,  x0, done      # if countdown == 0, goto done 
    sw   x2,  8(x0)         # write countdown to LED 
    addi x2,  x2, -1        # decrement 
    beq  x0,  x0, countdown # loop 

input: 
    lw   x2,  4(x0)         # read switch_port 
    beq  x2,  x0, main_loop # if 0, wait 
    sw   x2,  8(x0)         # display on LED 
    addi x10, x0, 1         # state = COUNTDOWN 
    beq  x0,  x0, main_loop # goto main_loop 

reset: 
    addi x10, x0, 0         # state = INPUT_WAITING 
    addi x2,  x0, 0         # clear countdown 
    sw   x0,  8(x0)         # clear LEDs 
    sw   x0,  0(x0)         # clear reset_flag 
    beq  x0,  x0, main_loop # goto main_loop 

done: 
    sw   x0,  8(x0)         # clear LEDs 
    beq  x0,  x0, main_loop # goto main_loop
