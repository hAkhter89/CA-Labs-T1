.text
.globl main

main:
    # LUI - load upper threshold
    lui  x1, 0x00001        # x1 = 0x00001000 = 4096

    # Read switches
    lw   x3, 4(x0)          # x3 = full 16-bit switch value

    # Extract lower 8 bits = number A
    andi x4, x3, 0xFF       # x4 = A = SW[7:0]

    # Extract upper 8 bits = number B
    srli x5, x3, 8          # x5 = B = SW[15:8]

    # JAL - call add subroutine
    jal  x6, add_sub        # x7 = A + B

back:
    # Store result to LED port
    sw   x7, 8(x0)          # display result on LEDs

    # BLT - if result < threshold, shift left by 2
    blt  x7, x1, shift      # if x7 < 4096, shift

    # Result >= threshold - halt
    sw   x7, 8(x0)          # display final value
    beq  x0, x0, done       # halt

shift:
    # shift result left by 2 (multiply by 4)
    slli x7, x7, 2          # x7 = x7 << 2
    sw   x7, 8(x0)          # display shifted result
    beq  x0, x0, done       # halt

done:
    beq  x0, x0, done       # infinite loop

# Add subroutine
add_sub:
    add  x7, x4, x5         # x7 = A + B
    jal  x6, back           # return
