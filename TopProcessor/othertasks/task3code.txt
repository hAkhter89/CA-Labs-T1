# ============================================
# TASK 3 - Count up loop using LUI, BLT, JAL (No JALR)
# ============================================
# Registers used:
# x1 = upper limit (from LUI)
# x2 = counter
# x6 = LED output register
# (x5 is unused here since we hardcode the return jump)

# INIT
lui  x1, 0x00001       # x1 = 0x00001000 (upper limit)
addi x2, x0, 0         # counter = 0

# MAIN LOOP
loop:
    jal  x0, display            # Jump down to display subroutine (use x0 as we aren't saving a return address)
    
return_here:                    # <--- EXPLICIT RETURN LABEL
    addi x2, x2, 1              # counter++ 
    blt  x2, x1, loop           # if counter < limit, loop back to top
    
    addi x2, x0, 0              # limit reached, reset counter to 0
    jal  x0, loop               # jump back to top

# DISPLAY SUBROUTINE
display:
    addi x6, x2, 0              # x6 = current counter (write to LEDs)
    jal  x0, return_here        # HARDCODED RETURN: Jump explicitly back to the increment phase
