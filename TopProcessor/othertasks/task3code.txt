.text
.globl main

main:
    # LUI - load upper bound
    lui x1, 0x00001        # x1 = 0x00001000 = 4096

    # Set up value to shift
    addi x2, x0, 1         # x2 = 1 (start value)

shift_loop:
    # Multiply by 2 via shift left
    slli x2, x2, 1         # x2 = x2 << 1 (x2 * 2)

    # JAL: jump to subroutine (saves return address in x5)
    jal  x5, display       

after_display:             
    # BLT - if x2 < upper bound, keep shifting
    blt  x2, x1, shift_loop 

done:
    # Store final value and halt
    sw   x2, 8(x0)         # display final value on LEDs
    beq  x0, x0, done      # infinite loop 

# Display subroutine
display:
    sw   x2, 8(x0)         # write to led_port
    
    # MANUAL RETURN: use jal to jump back and demonstrate blt
    jal  x0, after_display
